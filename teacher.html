📊 ตั้งค่าส่งคะแนนเข้า Google Sheets แบบ Realtime

เมื่อตั้งค่านี้ ทุกครั้งที่นักเรียนคนใดส่งคำตอบหรือทำแบบฝึกหัดเสร็จ คะแนนจะ เด้งเข้า Google Sheets ของครูทันทีแบบอัตโนมัติ โดยไม่ต้องส่งไฟล์! (ฟรี 100%)

วิธีทำง่ายๆ 3 ขั้นตอน:
  1. เปิด Google Sheets ใหม่ของคุณ แล้วไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  2. วางโค้ด Apps Script ด้านล่างนี้แทนที่ แล้วกด ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้รายการใหม่ (New Deployment) > ประเภทเว็บแอป (Web App) เลือกใครก็ได้ที่มีสิทธิ์เข้าถึง
  3. คัดลอก URL ของ Web App มาวางในช่องด้านล่าง แล้วกดบันทึก!
คลิกเพื่อดูโค้ด Google Apps Script (คัดลอกไปวางได้เลย) 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["วันเวลา", "ชื่อ-นามสกุล", "ชั้น", "เลขที่", "คะแนน XP", "ฉายา", "ภารกิจที่ผ่าน"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), data.name, data.grade, data.no, data.xp, data.badge, data.completedQuests]);
  return ContentService.createTextOutput("OK").setMimeType(ContentService.MimeType.TEXT);
}
